{{- /* Link render hook. Four cases: wiki: an Obsidian wikilink, rewritten by the prose partial absolute URL external link, opened in a new tab bare note an Obsidian relative link such as `bdd.md` or `some-note` anything else passed through unchanged Wikilinks and relative note links resolve to the post when it exists, and render as a "page does not exist" marker when it does not. */ -}} {{- $text := .Text -}} {{- $destination := .Destination -}} {{- $url := urls.Parse $destination -}} {{- $note := "" -}} {{- if strings.HasPrefix $destination "wiki:" -}} {{- $note = strings.TrimPrefix "wiki:" $destination -}} {{- else if and (not $url.IsAbs) (not (strings.HasPrefix $destination "/")) (not (strings.HasPrefix $destination "#")) (not (strings.HasPrefix $destination "mailto:")) -}} {{- $note = strings.TrimSuffix ".md" $destination -}} {{- end -}} {{- if $note -}} {{- with site.GetPage (printf "/posts/%s" $note) -}} {{ $text }} {{- else -}} {{- $label := printf "%s. Page does not exist." ($text | plainify) -}} {{ $text }} {{- end -}} {{- else -}} {{ $text }} {{- end -}} {{- /* chomp trailing newline */ -}}
